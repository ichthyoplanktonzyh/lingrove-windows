cmake_minimum_required(VERSION 3.24)
project(LingroveSpeechRuntime LANGUAGES C CXX)

# Speech's newer static libmtmd is isolated in this process. Completion/learning retain their
# own pinned LLamaSharp ABI and never load or resolve these libraries in the application process.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Prepare the exact pinned speech source and streaming patch first.")
endif()
# Even a direct CMake invocation must not label another library revision as the pinned runtime.
# Verification never fetches/applies/configures dependency input; preparation is an explicit step.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/prepare-speech-runtime.py"
  --source-dir "${LLAMA_SOURCE_DIR}" --verify-only RESULT_VARIABLE SPEECH_VERIFY_RESULT
  OUTPUT_QUIET ERROR_VARIABLE SPEECH_VERIFY_ERROR)
if(NOT SPEECH_VERIFY_RESULT EQUAL 0)
  message(FATAL_ERROR "Independent speech source verification failed: ${SPEECH_VERIFY_ERROR}")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" upstream EXCLUDE_FROM_ALL)
if(WIN32)
  set(SPEECH_PLATFORM SpeechPlatformWindows.cpp)
  set(SPEECH_PLATFORM_LIBS mfplat mfreadwrite mfuuid ole32 propsys)
elseif(APPLE)
  set(SPEECH_PLATFORM SpeechPlatformMac.cpp)
  set(SPEECH_PLATFORM_LIBS "-framework AudioToolbox" "-framework CoreFoundation")
else()
  message(FATAL_ERROR "Only the Windows adapter and Mac reference-side checks are supported.")
endif()
add_executable(lingrove-tts-helper SpeechHelper.cpp ${SPEECH_PLATFORM})
target_compile_features(lingrove-tts-helper PRIVATE cxx_std_17)
target_link_libraries(lingrove-tts-helper PRIVATE llama llama-common mtmd ${SPEECH_PLATFORM_LIBS})
if(MSVC)
  target_compile_options(lingrove-tts-helper PRIVATE /utf-8)
endif()
# Deliberately no install/copy-to-app/package target in this phase. Native CI checks the actual
# helper binary and platform effects without publishing an installer or downloading any weights.
add_executable(lingrove-tts-platform-check EXCLUDE_FROM_ALL SpeechPlatformChecks.cpp ${SPEECH_PLATFORM})
target_compile_features(lingrove-tts-platform-check PRIVATE cxx_std_17)
target_compile_definitions(lingrove-tts-platform-check PRIVATE LINGROVE_SPEECH_PLATFORM_CHECKS)
target_link_libraries(lingrove-tts-platform-check PRIVATE ${SPEECH_PLATFORM_LIBS})
if(MSVC)
  target_compile_options(lingrove-tts-platform-check PRIVATE /utf-8)
endif()
